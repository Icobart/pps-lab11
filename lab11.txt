%Ex1.1: search2
search2(Elem, List) :- append(_, [Elem, Elem | _], List).

%Ex1.2: search_two
search_two(Elem , List ) :- append(_, [Elem, _, Elem | _], List).

%Ex1.3: count
count([], E, 0).
count([E | T], E, N) :- count(T, E, N2), N is N2 + 1.
count([H | T], E, N) :- H \= E, count(T, E, N).

%Ex1.4: avg
sum([] , 0).
sum([H | T], N) :- sum (T , N2 ) , N is H + N2.
size(L , N) :- var (L) , !, generate (L , N).
size(L , N) :- length (L , N ).
avg([], 0).
avg(List, Avg) :- sum(List, Sum), size(List, Size), Avg is Sum / Size.

%Ex1.5: max and min
find_max(A, B, A) :- A >= B, !.
find_max(A, B, B) :- A < B.
find_min(A, B, A) :- A =< B, !.
find_min(A, B, B) :- A > B.
max([X], X, X).
max([H | T], Max, Min) :- max(T, MaxTail, MinTail), find_max(H, MaxTail, Max), find_min(H, MinTail, Min).

%Ex1.6: sublist
%split(List1, Elements, SubList1, SubList2) :- length(SubList1, Elements), append(SubList1, SubList2, List1).
split(List1, 0, [], List1).
split([H | T], Elements, [H | SubList1], SubList2) :- NewElements is Elements - 1, split(T, NewElements, SubList1, SubList2).

%Ex1.7: rotate
rotate([], []).
rotate([H | T], RotatedList) :- append(T, [H], RotatedList).

%Ex 1.8: dice
%dice(X) :- member(X, [1, 2, 3, 4, 5, 6]).
interval(A, B, A).
interval(A, B, X) :- A2 is A + 1, A2 =< B, interval(A2, B, X).
dice(X) :- interval(1, 6, X).

%Ex 1.9: three_dice
three_dice(L) :- dice(L1), dice(L2), dice(L3), L = [L1, L2, L3].

%Ex 2.1: distinct
distinct(List, DistinctList) :- distinct(List, DistinctList, []).
distinct([], [], _).
distinct([H | T], DistinctList, Seen) :- member(H, Seen), !, distinct(T, DistinctList, Seen).
distinct([H | T], [H | DistinctList], Seen) :- distinct(T, DistinctList, [H | Seen]).

%Ex2.3: other drops
dropFirst(X, [X | T], T) :- !.
dropFirst(X, [H | Xs], [H | L]) :- dropFirst(X, Xs, L).
dropLast(X, [H | Xs], [H | L]) :- dropLast(X, Xs, L), !.
dropLast(X, [X | T], T).
dropAll(_, [], []).
dropAll(X, [X | T], OutList) :- !, dropAll(X, T, OutList).
dropAll(X, [H | Xs], [H | L]) :- dropAll(X, Xs, L).

%Ex3.2: outDegree
outDegree([], _, 0).
outDegree([e(H1, _) | L], H1, Deg) :- !, outDegree(L, H1, DegTail), Deg is DegTail + 1.
outDegree([_ | L], Node, Deg) :- outDegree(L, Node, Deg).

%Ex3.3: reaching
reaching(Graph, Node, List) :- findall(Target, member(e(Node, Target), Graph), List).

%Ex3.4: nodes
extract_nodes([], []).
extract_nodes([e(H1, H2) | L], [H1, H2 | T]) :- extract_nodes(L, T).
nodes(Graph, Nodes) :- extract_nodes(Graph, NewNodes), distinct(NewNodes, Nodes).
